Atualiza a rotação do personagem para encarar o vetor de velocidade (vx, vz) function updateHumanRotation(agent) { // Atan2 calcula o ângulo entre o vetor (vx, vz) e a direção padrão do modelo const angle = Math.atan2(agent.vx, agent.vz); agent.human.root.rotation.y = angle; } // 2. Dentro da função updateAgents(), garanta que a rotação e colisão funcionem assim: function updateAgents() { let stepOccurred = false; pedestrians.forEach((agent) => { if (!isPaused) { // Move o agente agent.human.root.position.x += agent.vx; agent.human.root.position.z += agent.vz; const currX = agent.human.root.position.x; const currZ = agent.human.root.position.z; // Mantém o personagem SEMPRE olhando para a direção do movimento updateHumanRotation(agent); // Animação de passadas e balanço de braços agent.walkCycle += agent.speed * 2.8; agent.human.leftLeg.rotation.x = Math.sin(agent.walkCycle) * 0.6; agent.human.rightLeg.rotation.x = -Math.sin(agent.walkCycle) * 0.6; // Braços balançam opostos às pernas correspondentes agent.human.leftArm.rotation.x = -Math.sin(agent.walkCycle) * 0.5; agent.human.rightArm.rotation.x = Math.sin(agent.walkCycle) * 0.5; // Colisão com as bordas do piso: inverte a velocidade e aplica pequena variação angular if (Math.abs(currX) > HALF_FLOOR - 1.8) { agent.vx *= -1; // Pequeno desvio aleatório ao rebater para não andar sempre na mesma linha agent.vz += (Math.random() - 0.5) * 0.02; } if (Math.abs(currZ) > HALF_FLOOR - 1.8) { agent.vz *= -1; agent.vx += (Math.random() - 0.5) * 0.02; } // Re-normaliza a velocidade para manter a velocidade constante após a variação const currentSpeed = Math.hypot(agent.vx, agent.vz); if (currentSpeed > 0) { agent.vx = (agent.vx / currentSpeed) * agent.speed; agent.vz = (agent.vz / currentSpeed) * agent.speed; } // Adiciona o rastro de calor na malha addHeatAtWorldPos(currX, currZ); // Registro de passos para telemetria if (Math.hypot(currX - agent.lastStepX, currZ - agent.lastStepZ) > 1.2) { agent.lastStepX = currX; agent.lastStepZ = currZ; totalSteps++; totalJoules += JOULES_PER_STEP; stepOccurred = true; } } }); instantaneousWatts = stepOccurred ? (NUM_AGENTS * 6.5) + (Math.random() * 8.0) : Math.max(0, instantaneousWatts * 0.94); document.getElementById('val-joules').innerText = `${totalJoules.toFixed(1)} J`; document.getElementById('val-watts').innerText = `${instantaneousWatts.toFixed(1)} W`; document.getElementById('val-steps').innerText = totalSteps.toLocaleString('pt-BR'); }
