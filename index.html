// Paleta de gradiente térmico travada no Vermelho Intenso function getContinuousHeatColor(value) { // Normalização do valor térmico entre 0.0 e 1.0 const v = Math.min(1.0, Math.max(0.0, value / MAX_HEAT_VAL)); let r = 0, g = 0, b = 0; if (v < 0.20) { // Tráfego muito baixo: Azul escuro -> Ciano const t = v / 0.20; r = 0; g = Math.floor(180 * t); b = Math.floor(255 * t); } else if (v < 0.45) { // Tráfego baixo/médio: Ciano -> Verde const t = (v - 0.20) / 0.25; r = 0; g = Math.floor(180 + 75 * t); b = Math.floor(255 * (1 - t)); } else if (v < 0.70) { // Tráfego médio/alto: Verde -> Amarelo const t = (v - 0.45) / 0.25; r = Math.floor(255 * t); g = 255; b = 0; } else { // Tráfego máximo: Amarelo -> Vermelho Puro (Trava no Vermelho #FF0000) const t = (v - 0.70) / 0.30; r = 255; g = Math.floor(255 * (1 - t)); b = 0; } return [r, g, b]; }
