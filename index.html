// ========================================== // 7. MOTOR TÉRMICO DE ALTA PRECISÃO (SUAVE) // ========================================== // Aumentamos o limite para exigir mais passagens até atingir o vermelho máximo const MAX_HEAT_VAL = 8.0; // Paleta de gradiente térmico com curvas de transição suave function getContinuousHeatColor(value) { // Normalização do valor térmico entre 0.0 e 1.0 const v = Math.min(1.0, Math.max(0.0, value / MAX_HEAT_VAL)); let r = 0, g = 0, b = 0; if (v < 0.15) { // Calor muito baixo: Azul escuro / Ciano suave const t = v / 0.15; r = 0; g = Math.floor(120 * t); b = Math.floor(255 * t); } else if (v < 0.35) { // Calor baixo: Ciano -> Verde const t = (v - 0.15) / 0.20; r = 0; g = Math.floor(120 + 135 * t); b = Math.floor(255 * (1 - t)); } else if (v < 0.60) { // Calor médio: Verde -> Amarelo const t = (v - 0.35) / 0.25; r = Math.floor(255 * t); g = 255; b = 0; } else if (v < 0.85) { // Calor alto: Amarelo -> Laranja / Vermelho const t = (v - 0.60) / 0.25; r = 255; g = Math.floor(255 * (1 - t * 0.8)); b = 0; } else { // Ponto de saturação / Super quente: Vermelho -> Branco Térmico const t = (v - 0.85) / 0.15; r = 255; g = Math.floor(51 + 204 * t); b = Math.floor(255 * t); } return [r, g, b]; } // Aplicação de calor com atenuação ultra-suave function addHeatAtWorldPos(agentId, wx, wz) { const cx = Math.floor(((wx + HALF_FLOOR) / FLOOR_SIZE) * HEATMAP_RES); const cy = Math.floor(((wz + HALF_FLOOR) / FLOOR_SIZE) * HEATMAP_RES); // Raio do raio térmico levemente expandido para borramento gradual const radius = 22; const minX = Math.max(0, cx - radius); const maxX = Math.min(HEATMAP_RES - 1, cx + radius); const minY = Math.max(0, cy - radius); const maxY = Math.min(HEATMAP_RES - 1, cy + radius); const r2 = radius * radius; for (let y = minY; y <= maxY; y++) { for (let x = minX; x <= maxX; x++) { const dx = x - cx; const dy = y - cy; const dist2 = dx * dx + dy * dy; if (dist2 <= r2) { const dist = Math.sqrt(dist2); // Dissipação cossenoidal suave const falloff = 0.5 * (1 + Math.cos((dist / radius) * Math.PI)); const idx = y * HEATMAP_RES + x; // REDUZIDO: deposita menos calor por frame (de 0.045 para 0.008) heatMapData[idx] += falloff * 0.008; } } } } function updateHeatmapTexture() { const bgImgData = bgCtx.getImageData(0, 0, HEATMAP_RES, HEATMAP_RES); const data = bgImgData.data; let activePixels = 0; for (let i = 0; i < heatMapData.length; i++) { const val = heatMapData[i]; if (val > 0.02) { // Limiar mínimo para ignorar ruído activePixels++; const col = getContinuousHeatColor(val); const pIdx = i * 4; // Opacidade ajustada para ser bem transparente em pontos frios const alpha = Math.min(0.75, val * 0.25 + 0.1); data[pIdx] = Math.round(data[pIdx] * (1 - alpha) + col[0] * alpha); data[pIdx + 1] = Math.round(data[pIdx + 1] * (1 - alpha) + col[1] * alpha); data[pIdx + 2] = Math.round(data[pIdx + 2] * (1 - alpha) + col[2] * alpha); } } outputCtx.putImageData(bgImgData, 0, 0); floorTexture.needsUpdate = true; const coveragePercent = ((activePixels / heatMapData.length) * 100).toFixed(1); document.getElementById('val-coverage').innerText = `${coveragePercent}%`; }
